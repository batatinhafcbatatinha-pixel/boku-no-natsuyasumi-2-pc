cmake_minimum_required(VERSION 3.25)
project(Boku2Native LANGUAGES C CXX)

get_filename_component(WORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PS2X_RUNNER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/runner" CACHE PATH "Boku2 AOT sources" FORCE)
set(PS2X_GAME_CHUNK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/runner-chunks" CACHE PATH "Boku2 AOT chunks" FORCE)
set(PS2X_ENABLE_CODWAW_SUPPLEMENTS OFF CACHE BOOL "No other game's address bindings" FORCE)
# Diagnostic builds may enable the software GS (PS2X_GS=cpu) as a reference renderer.
option(BOKU2_ALLOW_CPU_GS "Allow PS2X_GS=cpu (reference rasterizer, diagnostics only)" OFF)
# macOS stops at OpenGL 4.1, so the GPU rasterizer (4.6 + fragment interlock) cannot run: the software GS is the renderer.
if(APPLE)
    set(BOKU2_SOFTWARE_GS_DEFAULT ON)
else()
    set(BOKU2_SOFTWARE_GS_DEFAULT OFF)
endif()
option(BOKU2_SOFTWARE_GS "Software GS only, no GPU scanout" ${BOKU2_SOFTWARE_GS_DEFAULT})
if(BOKU2_ALLOW_CPU_GS OR BOKU2_SOFTWARE_GS)
    set(PS2X_REQUIRE_GPU_RASTERIZER OFF CACHE BOOL "Boku2 diagnostic build: software GS allowed" FORCE)
else()
    set(PS2X_REQUIRE_GPU_RASTERIZER ON CACHE BOOL "Boku2 requires native host GPU rasterization" FORCE)
endif()
if(BOKU2_SOFTWARE_GS)
    set(PS2X_GPU_COMPUTE_SCANOUT OFF CACHE BOOL "No compute shaders on this host" FORCE)
    set(PS2X_GPU_TEXTURE_HANDOFF OFF CACHE BOOL "No shared GL texture on this host" FORCE)
else()
    set(PS2X_GPU_COMPUTE_SCANOUT ON CACHE BOOL "GPU detile, CRT blend and field scanout" FORCE)
    set(PS2X_GPU_TEXTURE_HANDOFF ON CACHE BOOL "Shared GL texture presentation without RGBA roundtrip" FORCE)
endif()
set(PS2X_ENABLE_LTCG OFF CACHE BOOL "Bound compilation memory" FORCE)
set(PS2X_FAST_LINK ON CACHE BOOL "Incremental native integration" FORCE)
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "No unused debug UI dependencies" FORCE)
set(PS2X_ENABLE_AGRESSIVE_LOGS OFF CACHE BOOL "No per-function logging" FORCE)
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "Use existing generator" FORCE)
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "Use existing analysis" FORCE)
set(PS2X_BUILD_TEST OFF CACHE BOOL "Do not run games/tests" FORCE)
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "Runtime only" FORCE)
set(BOKU2_PREBUILT_AOT "" CACHE FILEPATH "Existing verified Boku2 EE archive for runtime-only rebuilds")
if(BOKU2_PREBUILT_AOT AND NOT EXISTS "${BOKU2_PREBUILT_AOT}")
    message(FATAL_ERROR "BOKU2_PREBUILT_AOT does not exist")
endif()
set(PS2X_GAME_LIB_PREBUILT "${BOKU2_PREBUILT_AOT}" CACHE FILEPATH "Link only this game's generated EE code" FORCE)
if(WIN32)
    set(PS2X_FFMPEG_ROOT "${WORK_ROOT}/codwaw-ff/build/game/ThirdParty/ffmpeg-prefix/src/ffmpeg_external" CACHE PATH "Reuse installed decoder, no duplicate download")

    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Never download build dependencies" FORCE)
    set(FETCHCONTENT_SOURCE_DIR_RAYLIB "${WORK_ROOT}/codwaw-ff/build/game/_deps/raylib-src" CACHE PATH "Existing raylib checkout")
    if(NOT EXISTS "${FETCHCONTENT_SOURCE_DIR_RAYLIB}/CMakeLists.txt")
        message(FATAL_ERROR "Existing raylib checkout required; network acquisition is disabled")
    endif()
endif()
# Linux/other POSIX: FFmpeg from the system (pkg-config); raylib 5.5 fetched once unless
# FETCHCONTENT_SOURCE_DIR_RAYLIB points at an existing checkout.
set(PS2X_SPLIT_GAME_CODE ON CACHE BOOL "Generated code in ps2_game_code on every platform" FORCE)
# PS2Recomp runtime: the third_party/PS2Recomp submodule (fork, branch native-ports) when checked out,
# otherwise the shared checkout next to the game folders (../third_party/PS2Recomp).
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp/CMakeLists.txt")
    set(BOKU2_PS2RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp")
else()
    set(BOKU2_PS2RECOMP_DIR "${WORK_ROOT}/third_party/PS2Recomp")
endif()
get_filename_component(BOKU2_PS2RECOMP_DIR "${BOKU2_PS2RECOMP_DIR}" ABSOLUTE)
add_subdirectory("${BOKU2_PS2RECOMP_DIR}" ps2recomp)
include("${CMAKE_CURRENT_SOURCE_DIR}/runtime_memory.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/runtime_scheduler.cmake")
if(MSVC)
    # Debug info for the runtime and entry (not the AOT game library): cdb stacks with names. /Z7 works with sccache.
    target_compile_options(ps2_runtime PRIVATE /Z7)
    target_compile_options(ps2EntryRunner PRIVATE /Z7)
    target_link_options(ps2EntryRunner PRIVATE /DEBUG:FULL)
else()
    target_compile_options(ps2_runtime PRIVATE -g1)
    target_compile_options(ps2EntryRunner PRIVATE -g1)
endif()
get_target_property(BOKU2_ENTRY_SOURCES ps2EntryRunner SOURCES)
list(REMOVE_ITEM BOKU2_ENTRY_SOURCES "${BOKU2_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES ${BOKU2_ENTRY_SOURCES})
target_sources(ps2EntryRunner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
target_compile_definitions(ps2EntryRunner PRIVATE "PS2X_SHARED_MAIN_CPP=\"${BOKU2_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp\"")
# Functions outside the Ghidra map (scripts/gen_gap_functions.py), one unity unit; main.cpp fills empty table slots.
target_sources(ps2EntryRunner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/gap-functions.cpp")
if(MSVC)
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/gap-functions.cpp" PROPERTIES COMPILE_OPTIONS "/Od;/Ob0;/fp:precise;/bigobj")
else()
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/gap-functions.cpp" TARGET_DIRECTORY ps2EntryRunner
        PROPERTIES COMPILE_OPTIONS "-O0;-ffp-contract=off")
endif()

# Large resumable switches make whole-game optimizing compiles expensive.
# Bootstrap native code without optimizer expansion; optimize after coverage
# and dispatch integration stabilize. Other projects/runtime retain Release.
option(BOKU2_OPTIMIZE_AOT "Optimize the large generated game-code library" OFF)
if(MSVC)
    target_link_options(ps2EntryRunner PRIVATE /STACK:16777216 /MAP)
    if(BOKU2_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE /O1 /Ob1 /fp:precise)
    else()
        target_compile_options(ps2_game_code PRIVATE /Od /Ob0 /fp:precise)
    endif()
else()
    # GCC/Clang: same policy; no FMA contraction (the PS2 FPU has none). Game thread stack: PS2X_THREAD_STACK_MB.
    if(BOKU2_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE -O1 -ffp-contract=off)
    else()
        target_compile_options(ps2_game_code PRIVATE -O0 -ffp-contract=off)
    endif()
endif()

# Shared data installer: extracted files back all native EE/IOP optical reads.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tools/port-installer/runtime/extracted_files.cmake")
    set(PORT_INSTALLER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tools/port-installer" CACHE PATH "Shared installer checkout")
else()
    set(PORT_INSTALLER_DIR "${WORK_ROOT}/port-installer" CACHE PATH "Shared installer checkout")
endif()
include("${PORT_INSTALLER_DIR}/runtime/extracted_files.cmake")
